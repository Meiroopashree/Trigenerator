// File: Models/Astronaut.cs
namespace dotnetapp.Models
{
    public class Astronaut
    {
        public int AstronautID { get; set; } // Auto-incremented in DB
        public string Name { get; set; } // Astronaut's name
        public string Specialization { get; set; } // Astronaut's specialization
        public int ExperienceYears { get; set; } // Years of experience
        public int MissionCount { get; set; } // Number of missions
        public decimal TotalMissionHours { get; set; } // Calculated field (MissionCount * 200)
    }
}


using System;
using System.Data;
using System.Data.SqlClient;
using dotnetapp.Models;

namespace dotnetapp
{
    public static class ConnectionStringProvider
    {
        public static string ConnectionString { get; } = "User ID=sa;password=examlyMssql@123;server=localhost;Database=appdb;trusted_connection=false;Persist Security Info=False;Encrypt=False";
    }

    public class Program
    {
        static string connectionString = ConnectionStringProvider.ConnectionString;

        public static void Main(string[] args)
        {
            bool exit = false;
            while (!exit)
            {
                Console.WriteLine("\nSpace Mission Crew Management Menu");
                Console.WriteLine("1. Add Astronaut Record");
                Console.WriteLine("2. Display Limited Astronaut Records");
                Console.WriteLine("3. Update Astronaut Record");
                Console.WriteLine("4. Delete Astronaut Record");
                Console.WriteLine("5. Exit");
                Console.Write("Enter your choice (1-5): ");

                switch (Console.ReadLine())
                {
                    case "1":
                        Astronaut newAstronaut = new Astronaut
                        {
                            Name = ReadInput("Enter Astronaut Name: "),
                            Specialization = ReadInput("Enter Specialization: "),
                            ExperienceYears = int.Parse(ReadInput("Enter Experience Years: ")),
                            MissionCount = int.Parse(ReadInput("Enter Mission Count: "))
                        };

                        newAstronaut.TotalMissionHours = newAstronaut.MissionCount * 200; // Total hours calculation
                        AddAstronautRecord(newAstronaut);
                        break;

                    case "2":
                        int limit = int.Parse(ReadInput("Enter number of records to display: "));
                        DisplayLimitedAstronautRecords(limit);
                        break;

                    case "3":
                        string nameToUpdate = ReadInput("Enter Astronaut Name to update: ");
                        Astronaut updatedAstronaut = new Astronaut
                        {
                            Name = ReadInput("Enter New Name: "),
                            Specialization = ReadInput("Enter New Specialization: "),
                            ExperienceYears = int.Parse(ReadInput("Enter New Experience Years: ")),
                            MissionCount = int.Parse(ReadInput("Enter New Mission Count: "))
                        };

                        updatedAstronaut.TotalMissionHours = updatedAstronaut.MissionCount * 200;
                        UpdateAstronautRecord(nameToUpdate, updatedAstronaut);
                        break;

                    case "4":
                        string nameToDelete = ReadInput("Enter Astronaut Name to delete: ");
                        DeleteAstronautRecord(nameToDelete);
                        break;

                    case "5":
                        Console.WriteLine("Exiting the application...");
                        exit = true;
                        break;

                    default:
                        Console.WriteLine("Invalid choice.");
                        break;
                }
            }
        }

        private static string ReadInput(string prompt)
        {
            Console.Write(prompt);
            return Console.ReadLine();
        }

        public static void AddAstronautRecord(Astronaut astronaut)
        {
            if (astronaut.ExperienceYears < 0 || astronaut.MissionCount < 0)
            {
                Console.WriteLine("Experience years and mission count must be non-negative values.");
                return;
            }

            using (SqlConnection conn = new SqlConnection(connectionString))
            {
                SqlDataAdapter adapter = new SqlDataAdapter("SELECT * FROM Astronauts", conn);
                DataSet ds = new DataSet();
                adapter.Fill(ds, "Astronauts");

                DataTable table = ds.Tables["Astronauts"];
                DataRow newRow = table.NewRow();

                newRow["Name"] = astronaut.Name;
                newRow["Specialization"] = astronaut.Specialization;
                newRow["ExperienceYears"] = astronaut.ExperienceYears;
                newRow["MissionCount"] = astronaut.MissionCount;
                newRow["TotalMissionHours"] = astronaut.TotalMissionHours;

                table.Rows.Add(newRow);

                SqlCommandBuilder commandBuilder = new SqlCommandBuilder(adapter);
                adapter.Update(ds, "Astronauts");

                Console.WriteLine("Astronaut record added successfully.");
            }
        }

        public static void DisplayLimitedAstronautRecords(int limit)
        {
            using (SqlConnection conn = new SqlConnection(connectionString))
            {
                SqlDataAdapter adapter = new SqlDataAdapter($"SELECT TOP {limit} * FROM Astronauts ORDER BY TotalMissionHours DESC", conn);
                DataSet ds = new DataSet();
                adapter.Fill(ds, "Astronauts");

                if (ds.Tables["Astronauts"].Rows.Count > 0)
                {
                    foreach (DataRow row in ds.Tables["Astronauts"].Rows)
                    {
                        Console.WriteLine($"Name: {row["Name"]}, Specialization: {row["Specialization"]}, Experience: {row["ExperienceYears"]} years, Missions: {row["MissionCount"]}, Total Hours: {row["TotalMissionHours"]}");
                    }
                }
                else
                {
                    Console.WriteLine("No astronaut records found.");
                }
            }
        }

        public static void UpdateAstronautRecord(string name, Astronaut updatedAstronaut)
        {
            if (updatedAstronaut.ExperienceYears < 0 || updatedAstronaut.MissionCount < 0)
            {
                Console.WriteLine("Experience years and mission count must be non-negative values.");
                return;
            }

            using (SqlConnection conn = new SqlConnection(connectionString))
            {
                SqlDataAdapter adapter = new SqlDataAdapter("SELECT * FROM Astronauts", conn);
                DataSet ds = new DataSet();
                adapter.Fill(ds, "Astronauts");

                DataRow[] rowsToUpdate = ds.Tables["Astronauts"].Select($"Name = '{name}'");

                if (rowsToUpdate.Length > 0)
                {
                    DataRow row = rowsToUpdate[0];
                    row["Name"] = updatedAstronaut.Name;
                    row["Specialization"] = updatedAstronaut.Specialization;
                    row["ExperienceYears"] = updatedAstronaut.ExperienceYears;
                    row["MissionCount"] = updatedAstronaut.MissionCount;
                    row["TotalMissionHours"] = updatedAstronaut.TotalMissionHours;

                    SqlCommandBuilder commandBuilder = new SqlCommandBuilder(adapter);
                    adapter.Update(ds, "Astronauts");

                    Console.WriteLine("Astronaut record updated successfully.");
                }
                else
                {
                    Console.WriteLine("No matching astronaut record found.");
                }
            }
        }

        public static void DeleteAstronautRecord(string name)
        {
            using (SqlConnection conn = new SqlConnection(connectionString))
            {
                SqlDataAdapter adapter = new SqlDataAdapter("SELECT * FROM Astronauts", conn);
                DataSet ds = new DataSet();
                adapter.Fill(ds, "Astronauts");

                DataRow[] rowsToDelete = ds.Tables["Astronauts"].Select($"Name = '{name}'");

                if (rowsToDelete.Length > 0)
                {
                    rowsToDelete[0].Delete();

                    SqlCommandBuilder commandBuilder = new SqlCommandBuilder(adapter);
                    adapter.Update(ds, "Astronauts");

                    Console.WriteLine("Astronaut record deleted successfully.");
                }
                else
                {
                    Console.WriteLine("No matching astronaut record found.");
                }
            }
        }
    }
}

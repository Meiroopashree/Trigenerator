   private void InsertAstronautIntoDatabase(string name, string specialization, decimal experienceYears, int missionCount, decimal totalMissionHours)
{
    using (SqlConnection conn = new SqlConnection(connectionString))
    {
        // Open connection
        conn.Open();

        // Create a DataAdapter to fetch the schema of the Astronauts table
        SqlDataAdapter adapter = new SqlDataAdapter("SELECT * FROM Astronauts", conn);
        SqlCommandBuilder commandBuilder = new SqlCommandBuilder(adapter);

        // Fill a DataSet with the Astronauts table structure
        DataSet dataSet = new DataSet();
        adapter.Fill(dataSet, "Astronauts");

        // Access the DataTable from the DataSet
        DataTable astronautsTable = dataSet.Tables["Astronauts"];

        // Create a new DataRow and populate it with data
        DataRow newRow = astronautsTable.NewRow();
        newRow["Name"] = name;
        newRow["Specialization"] = specialization;
        newRow["ExperienceYears"] = experienceYears;
        newRow["MissionCount"] = missionCount;
        newRow["TotalMissionHours"] = totalMissionHours;


        // Add the new DataRow to the DataTable
        astronautsTable.Rows.Add(newRow);

        // Use the DataAdapter to update the database with changes made in the DataSet
        adapter.Update(dataSet, "Astronauts");

        Console.WriteLine("Astronaut record inserted successfully using disconnected architecture.");
    }
}


        private string CaptureConsoleOutput(Action action)
        {
            consoleOutput.GetStringBuilder().Clear();
            action.Invoke();
            return consoleOutput.ToString();
        }
